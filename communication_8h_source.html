153 std::pair<int, int> removeTransactionFromQueue(const AddressType &address)
154 {
155 xSemaphoreTake(addressHashTableMutex, portMAX_DELAY);
156 // Check if the address exists in the hash table
157 auto it = addressHashTable.find(address.toString());
158 if (it != addressHashTable.end())
159 {
160 // If the address exists, retrieve the queue associated with it
161 std::queue<std::pair<uint16_t, uint16_t>> &queue = it->second;
162 // Check if the queue is not empty
163 if (!queue.empty())
164 {
165 // Get the next value in line from the queue
166 std::pair<uint16_t, uint16_t> transaction = queue.front();
167 // Remove the value from the queue
168 queue.pop();
169 xSemaphoreGive(addressHashTableMutex);
170 // Return the transaction ID and unit identifier
171 return {transaction.first, transaction.second};
172 }
173 }
174 xSemaphoreGive(addressHashTableMutex);
175 return {-1, -1}; // Could not be found or the queue is empty
176 }

226 MBAP createTCPIPPacket(PDU &frame, int transaction_id = -1, int unit_identifier = 0xFF)
227 {
228 MBAP packet;
229 uint16_t unique_number;
230
231 if (transaction_id == -1)
232 {
233 std::uniform_int_distribution<> distribution(0, 0xFFFF);
234 unique_number = distribution(gen);
235 }
236 else

238 unique_number = transaction_id;
239 }
240
241 packet.transaction_id = unique_number;
242 packet.protocol_id = 0x00; // Modbus protocol id
243 packet.length_field = 0x01 + 0x01 + frame.data.size();
244 packet.unit_identifier =
245 unit_identifier; // On TCP/IP, the MODBUS server is addressed using its IP address;
246 // therefore, the MODBUS Unit Identifier is useless. The value 0xFF
247 // has to be used.
248 packet.pdu = frame;
