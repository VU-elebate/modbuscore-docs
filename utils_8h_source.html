177struct MBAP {
178 uint16_t transaction_id;
179 uint16_t protocol_id;
180 uint16_t length_field;
181 uint8_t unit_identifier;
182 PDU pdu;
183
189 std::vector<uint8_t> serializeMBAP(void) {
190 std::vector<uint8_t> serializedData;
191
192 // Serialize transaction id (16 bits)
193 serializedData.push_back(transaction_id & 0xFF);
194 serializedData.push_back((transaction_id >> 8) & 0xFF);
195
196 // Serialize protocol id (16 bits)
197 serializedData.push_back(protocol_id & 0xFF);
198 serializedData.push_back((protocol_id >> 8) & 0xFF);
199
200 // Serialize length field (16 bits)
201 serializedData.push_back(length_field & 0xFF00);
202 serializedData.push_back(length_field & 0x00FF);
203
204 // Serialize unit identifier (8 bits)
205 serializedData.push_back(unit_identifier);
206
207 // Serialize PDU
208 std::vector<uint8_t> serializedPDU = pdu.serializePDU();
209
210 // Append serialized PDU to serializedData
211 serializedData.insert(serializedData.end(), serializedPDU.begin(), serializedPDU.end());
212
213 return serializedData;
214 }
215
222 MBAP MBAPFromSerializedData(const std::vector<uint8_t> &serializedData) {
223 MBAP mbap;
224
225 // Extract transaction id
226 mbap.transaction_id = serializedData[0] | (serializedData[1] << 8);
227 // Extract protocol id
228 mbap.protocol_id = serializedData[2] | (serializedData[3] << 8);
229 // Extract field length
230 mbap.length_field = serializedData[4] | (serializedData[5] << 8);
231 // Extract unit identifier
232 mbap.unit_identifier = serializedData[6];
233 // Create PDU from serialized data
234 std::vector<uint8_t> pduSerializedData(serializedData.begin() + 7, serializedData.end());
235
236 mbap.pdu = mbap.pdu.PDUFromSerializedData(pduSerializedData);
237
238 return mbap;
239 }
